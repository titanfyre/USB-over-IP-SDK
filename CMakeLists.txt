cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(USBOverIP VERSION 0.1.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Build with Visual Studio C++ on Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(TRACE_ASAN "AddressSanitizer user-mode tests" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(TRACE_ASAN)
 add_compile_options(/fsanitize=address /Zi)
 add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(USBOverIP SDK/USBOverIP.cpp SDK/Transport.cpp SDK/License.cpp)
target_link_libraries(USBOverIP PUBLIC winusb setupapi cfgmgr32 ole32 ws2_32 secur32 crypt32 ncrypt bcrypt winhttp advapi32)
add_executable(USBOverIPStudio WIN32 App/Main.cpp)
target_link_libraries(USBOverIPStudio PRIVATE USBOverIP user32 gdi32 comctl32 shell32)
target_link_options(USBOverIPStudio PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(USBOverIPCLI App/Console.cpp)
target_link_libraries(USBOverIPCLI PRIVATE USBOverIP)
target_link_options(USBOverIPCLI PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(USBOverIPExample Examples/connect.cpp)
target_link_libraries(USBOverIPExample PRIVATE USBOverIP)
target_link_options(USBOverIPExample PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
add_executable(USBOverIPTests Tests/Protocol.cpp)
target_link_libraries(USBOverIPTests PRIVATE USBOverIP)
enable_testing()
add_test(NAME USBProtocolAndTLS COMMAND USBOverIPTests)
add_library(USBOverIPSetupCore InstallerUI/Setup.cpp InstallerUI/RuntimeSetup.cpp)
target_include_directories(USBOverIPSetupCore PRIVATE SDK)
target_link_libraries(USBOverIPSetupCore PUBLIC setupapi newdev wintrust crypt32 advapi32 ole32)
add_executable(USBOverIPInstallerTests Tests/Installer.cpp)
target_link_libraries(USBOverIPInstallerTests PRIVATE USBOverIPSetupCore)
add_test(NAME InstallerIdentityAndSignatureRefusal COMMAND USBOverIPInstallerTests)
set(RUNTIME_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed Titan SDK Runtime package")
set(DRIVER_PACKAGE_DIRECTORY "" CACHE PATH "Opaque Microsoft-signed Titan USB-over-IP package")
set(PACKAGE_RESOURCE "")
if(RUNTIME_PACKAGE_DIRECTORY OR DRIVER_PACKAGE_DIRECTORY)
 enable_language(RC)
 set(PACKAGE_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/DriverPackage.rc")
 file(WRITE "${PACKAGE_RESOURCE}" "")
 foreach(package RUNTIME DRIVER)
  if(${package}_PACKAGE_DIRECTORY)
   if(package STREQUAL "RUNTIME")
    set(service TitanSdkRuntime)
    set(id 101)
   else()
    set(service TitanUSBOverIP)
    set(id 111)
   endif()
   foreach(extension inf cat sys)
    if(NOT EXISTS "${${package}_PACKAGE_DIRECTORY}/${service}.${extension}")
     message(FATAL_ERROR "Incomplete opaque signed package")
    endif()
    file(APPEND "${PACKAGE_RESOURCE}" "${id} RCDATA \"${${package}_PACKAGE_DIRECTORY}/${service}.${extension}\"\n")
    math(EXPR id "${id}+1")
   endforeach()
  endif()
 endforeach()
endif()
add_executable(USBOverIPSetup WIN32 InstallerUI/Main.cpp ${PACKAGE_RESOURCE})
add_executable(USBOverIPSetupConsole InstallerConsole/Main.cpp ${PACKAGE_RESOURCE})
foreach(target USBOverIPSetup USBOverIPSetupConsole)
 target_link_libraries(${target} PRIVATE USBOverIPSetupCore user32 gdi32)
 target_link_options(${target} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
foreach(target USBOverIPStudio USBOverIPCLI USBOverIPExample USBOverIPTests USBOverIPInstallerTests USBOverIPSetup USBOverIPSetupConsole)
 set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
